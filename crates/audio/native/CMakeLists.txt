cmake_minimum_required(VERSION 3.20)
project(bumblebee_speech_wrapper LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Bumblebee native audio requires a 64-bit target")
endif()
if(NOT DEFINED SPEECH_SDK_ROOT)
  message(FATAL_ERROR "SPEECH_SDK_ROOT must point to the downloaded SDK")
endif()
add_library(bumblebee_speech_wrapper SHARED speech_wrapper.cpp)
target_compile_features(bumblebee_speech_wrapper PRIVATE cxx_std_17)
target_include_directories(bumblebee_speech_wrapper PRIVATE "${SPEECH_SDK_ROOT}/include/cxx_api" "${SPEECH_SDK_ROOT}/include/c_api")
if(WIN32)
  set_target_properties(bumblebee_speech_wrapper PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
  target_link_libraries(bumblebee_speech_wrapper PRIVATE "${SPEECH_SDK_ROOT}/lib/x64/Microsoft.CognitiveServices.Speech.core.lib")
else()
  find_package(Threads REQUIRED)
  target_link_libraries(bumblebee_speech_wrapper PRIVATE "${SPEECH_SDK_ROOT}/lib/x64/libMicrosoft.CognitiveServices.Speech.core.so" Threads::Threads)
  set_target_properties(bumblebee_speech_wrapper PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS bumblebee_speech_wrapper RUNTIME DESTINATION . LIBRARY DESTINATION .)
